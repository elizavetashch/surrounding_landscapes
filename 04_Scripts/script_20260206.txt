imp_num  <- function(x) { x[is.na(x)] <- median(x, na.rm = TRUE); x }
imp_fact <- function(x) { tab <- table(x, useNA = "no"); if (length(tab)) x[is.na(x)] <- names(which.max(tab)); x }
is_const <- function(x) {
  if (is.numeric(x)) sd(x, na.rm = TRUE) == 0 else length(na.omit(unique(x))) <= 1
}

make_surface_shannon_field <- function(model, shan_var, field_var,
                                         buffer_label,
                                         n_shan = 60, n_field = 60) {
    mf <- model.frame(model)          # data actually used
    response <- all.vars(formula(model))[1]
    
    # sequences over observed ranges
    shan_seq  <- seq(min(mf[[shan_var]],  na.rm = TRUE),
                     max(mf[[shan_var]],  na.rm = TRUE),
                     length.out = n_shan)
    field_seq <- seq(min(mf[[field_var]], na.rm = TRUE),
                     max(mf[[field_var]], na.rm = TRUE),
                     length.out = n_field)
    
    grid <- expand.grid(
      shan  = shan_seq,
      field = field_seq
    )
    
    # map to variable names used in the model
    grid[[shan_var]]  <- grid$shan
    grid[[field_var]] <- grid$field
    
    # set all other predictors to typical values
    other_vars <- setdiff(names(mf), c(response, shan_var, field_var))
    for (v in other_vars) {
      if (is.numeric(mf[[v]])) {
        grid[[v]] <- mean(mf[[v]], na.rm = TRUE)
      } else if (is.factor(mf[[v]])) {
        grid[[v]] <- factor(levels(mf[[v]])[1], levels = levels(mf[[v]]))
      } else {
        grid[[v]] <- mf[[v]][1]
      }
    }
    
    # predict (fixed effects only)
    grid$pred <- as.numeric(predict(model, newdata = grid, re.form = NA))
    
    # convenience columns
    grid$x       <- grid[[shan_var]]
    grid$y       <- grid[[field_var]]
    grid$buffer  <- buffer_label
    
    grid
  }
  
  surf_1000 <- make_surface_shannon_field(
    model        = shannon1000.h1,
    shan_var     = "shannon.1000",
    field_var    = "fieldsize.centre.numeric",
    buffer_label = "1 km"
  )
  
  surf_2500 <- make_surface_shannon_field(
    model        = shannon2500.h1,
    shan_var     = "shannon.2500",
    field_var    = "fieldsize.centre.numeric",
    buffer_label = "2.5 km"
  )
  
  surf_5000 <- make_surface_shannon_field(
    model        = shannon5000.h1,
    shan_var     = "shannon.5000",
    field_var    = "fieldsize.centre.numeric",
    buffer_label = "5 km"
  )
  
  surf_all <- bind_rows(surf_1000, surf_2500, surf_5000) %>%
    mutate(buffer = factor(buffer, levels = c("1 km", "2.5 km", "5 km")))
  
  ggplot(surf_all, aes(x = x, y = y)) +
    geom_tile(aes(fill = pred)) +
    stat_contour(aes(z = pred),
                 colour = "white",
                 bins = 10,
                 size = 0.3,
                 alpha = 0.8) +
    facet_wrap(~ buffer, nrow = 1, scales = "free") +
    scale_fill_gradient(
      low  = "#4C9BD7",   # bluish (lower / more negative lrr)
      high = "#4B0055",   # dark magenta/purple (higher / more positive lrr)
      name = "Predicted lrr"
    ) +
    labs(
      x = "Shannon landscape diversity",
      y = "Field size (ha)",
      fill = "Predicted lrr",
      title = "Interaction surface: Shannon diversity × field size") +
    theme_minimal(base_size = 13) +
    theme(
      panel.grid    = element_blank(),
      strip.text    = element_text(face = "bold"),
      plot.title    = element_text(face = "bold", hjust = 0),
      plot.subtitle = element_text(hjust = 0),
      panel.border  = element_rect(colour = "black", fill = NA, linewidth = 0.4),
      panel.spacing = unit(0.8, "lines"))

make_surface <- function(model, dat, shannon_var, poll_label, buffer_label,
                         n_field = 50, n_shannon = 50) {
  # valid combo of factor levels etc.
  mf   <- model.frame(model)
  base <- mf[1, , drop = FALSE]
  
  # sequences over observed ranges in the respective subset
  fs_seq <- seq(
    min(dat$fieldsize.centre.numeric, na.rm = TRUE),
    max(dat$fieldsize.centre.numeric, na.rm = TRUE),
    length.out = n_field
  )
  sh_seq <- seq(
    min(dat[[shannon_var]], na.rm = TRUE),
    max(dat[[shannon_var]], na.rm = TRUE),
    length.out = n_shannon
  )
  
  grid <- expand.grid(
    fieldsize.centre.numeric = fs_seq,
    sh = sh_seq
  )
  
  # build newdata from base row, overwrite focal vars
  newdata <- base[rep(1, nrow(grid)), , drop = FALSE]
  newdata$fieldsize.centre.numeric <- grid$fieldsize.centre.numeric
  newdata[[shannon_var]] <- grid$sh
  
  # fixed-effects predictions
  grid$pred <- predict(model, newdata = newdata, re.form = NA)
  grid$poll <- poll_label
  grid$buffer <- buffer_label
  names(grid)[names(grid) == "sh"] <- "shannon"
  
  grid
}

make_curves_nat_treat_crop <- function(model, nat_var, buffer_label,
                                       treat_var = "treatment",
                                       crop_var  = "crop.type.orig.grouped.big",
                                       n_nat = 60) {
  mf <- model.frame(model)   # data used in model
  response <- all.vars(formula(model))[1]
  
  # range of nat.hab in this model
  nat_seq <- seq(
    min(mf[[nat_var]], na.rm = TRUE),
    max(mf[[nat_var]], na.rm = TRUE),
    length.out = n_nat
  )
  
  treat_levs <- levels(mf[[treat_var]])
  crop_levs  <- levels(mf[[crop_var]])
  
  # all treatment × crop combinations
  combos <- expand.grid(
    trt  = treat_levs,
    crop = crop_levs,
    KEEP.OUT.ATTRS = FALSE,
    stringsAsFactors = FALSE
  )
  
  base <- mf[1, , drop = FALSE]            # valid levels for factors etc.
  other_vars <- setdiff(names(mf), c(response, nat_var, treat_var, crop_var))
  
  res_list <- lapply(seq_len(nrow(combos)), function(i) {
    # prediction grid over nat for this treatment × crop combo
    nd <- base[rep(1, length(nat_seq)), , drop = FALSE]
    
    nd[[nat_var]]   <- nat_seq
    nd[[treat_var]] <- factor(combos$trt[i],  levels = treat_levs)
    nd[[crop_var]]  <- factor(combos$crop[i], levels = crop_levs)
    
    # typical values for other predictors
    for (v in other_vars) {
      if (is.numeric(mf[[v]])) {
        nd[[v]] <- mean(mf[[v]], na.rm = TRUE)
      } else if (is.factor(mf[[v]])) {
        nd[[v]] <- factor(levels(mf[[v]])[1], levels = levels(mf[[v]]))
      } else {
        nd[[v]] <- mf[[v]][1]
      }
    }
    
    # predicted lrr (fixed effects only; quadratic is inside model)
    pred <- as.numeric(predict(model, newdata = nd, re.form = NA))
    
    data.frame(
      nat       = nat_seq,
      pred      = pred,
      treatment = combos$trt[i],
      crop      = combos$crop[i],
      buffer    = buffer_label,
      stringsAsFactors = FALSE
    )
  })
  
  do.call(rbind, res_list)
}

# Plot results
make_surface <- function(model, nat_var, mod_var, n = 50) {
  # model: an lmer model
  # nat_var: character name of natural habitat variable in model/data
  # mod_var: character name of interacting variable (climate or richness)
  # n: number of grid steps per axis
  mf <- model.frame(model)  # data actually used in the model
  # response name (first variable in formula)
  response <- all.vars(formula(model))[1]
  # sequences over observed ranges
  nat_seq <- seq(min(mf[[nat_var]], na.rm = TRUE),
                 max(mf[[nat_var]], na.rm = TRUE),
                 length.out = n)
  mod_seq <- seq(min(mf[[mod_var]], na.rm = TRUE),
                 max(mf[[mod_var]], na.rm = TRUE),
                 length.out = n)
  # base grid for the 2D surface
  grid <- expand.grid(
    nat = nat_seq,
    mod = mod_seq)
  names(grid) <- c(nat_var, mod_var)
  # fill in all other predictors with "typical" values
  other_vars <- setdiff(names(mf), c(response, nat_var, mod_var))
  for (v in other_vars) {
    if (is.numeric(mf[[v]])) {
      grid[[v]] <- mean(mf[[v]], na.rm = TRUE)
    } else if (is.factor(mf[[v]])) {
      grid[[v]] <- factor(levels(mf[[v]])[1], levels = levels(mf[[v]]))
    } else {
      grid[[v]] <- mf[[v]][1]}}
  # predict lrr on the grid (fixed effects only: re.form = NA)
  grid$pred <- as.numeric(predict(model, newdata = grid, re.form = NA))
  grid$x <- grid[[nat_var]]
  grid$y <- grid[[mod_var]]
  grid}

# Moran's I test for all models
###############################
# install.packages("spdep")   # if not installed
library(spdep)
library(dplyr)
run_moran_for_model <- function(model, df, k = 4, model_name = deparse(substitute(model))) {
  # data actually used in model
  mf  <- model.frame(model)
  res <- resid(model)
  # use rownames to link back to df (since data = df in lmer())
  idx <- as.integer(rownames(mf))
  lon <- df$longitude.decimal[idx]
  lat <- df$latitude.decimal[idx]
  # keep only complete cases (coords + residual)
  ok <- complete.cases(lon, lat, res)
  lon <- lon[ok]
  lat <- lat[ok]
  res <- res[ok]
  coords <- cbind(lon, lat)
  # spatial weights: k-nearest neighbours (k = 4 by default)
  knn <- knearneigh(coords, k = k)
  nb  <- knn2nb(knn)
  lw  <- nb2listw(nb, style = "W")
  mt <- moran.test(res, lw)
  data.frame(
    model       = model_name,
    moran_I     = unname(mt$estimate[["Moran I statistic"]]),
    expectation = unname(mt$estimate[["Expectation"]]),
    variance    = unname(mt$estimate[["Variance"]]),
    p_value     = mt$p.value,
    stringsAsFactors = FALSE)}

model_names <- c("shannon1000.h1", "shannon2500.h1", "shannon5000.h1",
  "shannon1000.h1.nopoll", "shannon2500.h1.nopoll", "shannon5000.h1.nopoll",
  "shannon1000.h1.poll",   "shannon2500.h1.poll",   "shannon5000.h1.poll",
  "croptreat1000.h2", "croptreat2500.h2", "croptreat5000.h2",
  "srclim1000.h3", "srclim2500.h3", "srclim5000.h3",
  "crop1000.h4", "crop2500.h4", "crop5000.h4",
  "landsc1000.h5", "landsc2500.h5", "landsc5000.h5")

# grab the objects from the environment
models <- mget(model_names)
moran_results <- bind_rows(lapply(names(models), function(nm) {
    run_moran_for_model(models[[nm]], df = df, k = 4, model_name = nm)}))
moran_results

write.csv(moran_results, file = "moran_test_results.csv")
