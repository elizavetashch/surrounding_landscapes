# Source: https://rfunctions.blogspot.com/2017/08/extracting-data-from-rasters-using.html
options(scipen = 999)

install.packages("raster")
library(raster)

# Use "raster" function to load our raster into R.
sr <- raster(choose.files())
target_crs <- "+proj=longlat +datum=WGS84" # Set WGS84
sr <- projectRaster(sr, crs = target_crs, 
                    res = c(0.1, 0.1),  # Set desired resolution
                    ext = extent(-190, 190, -90, 90)) 

# Now we can plot our raster data:
plot(sr, main="SR", xlab="Longitude",ylab="Latitude",cex.axis=1.3, cex.lab=1.4, cex.main=1.5,col=colorRampPalette(c("darkseagreen1", "darkgreen"))(50))

# Load the table into R
places<-read.csv(choose.files(), head=TRUE, sep=",")

# Let us now organize coordinates into a spatial object:
coords<-data.frame(lon=places$longitude_decimal, lat=places$latitude_decimal)
coordinates(coords)<-c("lon","lat")

library(maps)
map()
points(coords, pch=16)

# So, let us use "extract" function to get values for our coordinates. Observe that "x" argument corresponds to the 
# raster we are using (in our example, annual mean temperature), and "y" argument corresponds to the spatial 
# object with coordinates of places for which we want temperature values.

val<-extract(x=sr, y=coords)

# Now let's download temperature and precipitation data
library(sp)

# Time period: 1970-2000
temp_data <- worldclim_global(var="tavg", res=10, path=tempdir())
prec_data <- worldclim_global(var = "prec", res = 10, path = tempdir())

# Convert coordinates to a format that the function can understand
coords_vect <- vect(coords, crs = "EPSG:4326")

# 
temp_values <- extract(temp_data, coords_vect)
prec_values <- extract(prec_data, coords_vect)
temp_avg <- subset(temp_values, select = -c(ID))
prec_avg <- subset(prec_values, select = -c(ID))

# Calculate the mean of all columns per row
temp_avg <- rowMeans(temp_avg, na.rm = FALSE)
prec_avg <- rowMeans(prec_avg, na.rm = FALSE)


result <- cbind(coords_vect, temp_avg, prec_avg)
df3 <- cbind(df3, temp_avg, prec_avg)
write.csv(df3, file = "df3.csv")
