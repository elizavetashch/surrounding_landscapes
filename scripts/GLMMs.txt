# Let's start by comparing the three buffers. Which buffer explains best the response ratio?

scale_df <- function(df) {
  numeric_cols <- sapply(df, is.numeric)
  df[numeric_cols] <- lapply(df[numeric_cols], scale)
  return(df)}
df <- scale_df(df)
